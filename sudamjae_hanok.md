    // Simple Interactive Reservation Price Calculator
    const nightsSelect = document.getElementById('stay-nights');
    const guestSelect = document.getElementById('guest-count');
    const priceDisplay = document.getElementById('calculated-price');

    function updatePrice() {
      if (!nightsSelect || !guestSelect || !priceDisplay) return;
      const baseNightRate = 750000;
      const nights = parseInt(nightsSelect.value, 10);
      const guests = parseInt(guestSelect.value, 10);
      
      const extraGuestFee = Math.max(0, guests - 4) * 50000;
      
      // Consecutive nights discounts
      let discountRate = 1.0;
      if (nights === 2) discountRate = 0.90;
      if (nights === 3) discountRate = 0.85;

      const total = Math.round(((baseNightRate * nights * discountRate) + (extraGuestFee * nights)));
      priceDisplay.innerText = '₩ ' + total.toLocaleString();
    }

    if (nightsSelect && guestSelect) {
      nightsSelect.addEventListener('change', updatePrice);
      guestSelect.addEventListener('change', updatePrice);
    }

    // Lightbox Gallery Modal Handler
    function openGalleryModal(src, title, desc) {
      const modal = document.getElementById('gallery-modal');
      const img = document.getElementById('modal-image');
      const titleEl = document.getElementById('modal-title');
      const descEl = document.getElementById('modal-desc');

      if (modal && img && titleEl && descEl) {
        img.src = src;
        img.alt = title;
        titleEl.innerText = title;
        descEl.innerText = desc;
        modal.classList.remove('hidden');
        modal.classList.add('flex');
        document.body.style.overflow = 'hidden';
      }
    }

    function closeGalleryModal() {
      const modal = document.getElementById('gallery-modal');
      if (modal) {
        modal.classList.add('hidden');
        modal.classList.remove('flex');
        document.body.style.overflow = '';
      }
    }

    // Close modal on escape key press
    document.addEventListener('keydown', (e) => {
      if (e.key === 'Escape') closeGalleryModal();
    });
